{{- /* The HTML elements that read the same in any Markdown: drops scripts, styles and SVG icons, turns embeds into links, and turns links, headings, images and emphasis into Markdown. Used on its own for raw HTML inside Markdown content, and as the first step of from-html. Leaves block structure and whitespace alone. Returns a string. */ -}} {{- $h := string . -}} {{- $h = $h | replaceRE `(?s)` "" | replaceRE `<\?xml[^>]*\?>` "" -}} {{- range slice "script" "style" "svg" "template" "button" "noscript" "label" "form" -}} {{- $h = $h | replaceRE (printf `(?s)<%s\b.*?` . .) "" -}} {{- end -}} {{- $h = $h | replaceRE `]*>` "" -}} {{- /* Embedded players (YouTube, slides, legacy Flash) become a link to what they embed. */ -}} {{- $h = $h | replaceRE `(?s)]*?\bsrc="(?:https?:)?//(?:www\.)?youtube(?:-nocookie)?\.com/embed/([A-Za-z0-9_-]+)[^"]*"[^>]*>.*?` "Video: [https://www.youtube.com/watch?v=$1](https://www.youtube.com/watch?v=$1)" -}} {{- $h = $h | replaceRE `(?s)]*?\bsrc="([^"]+)"[^>]*>.*?` "Embedded: [$1]($1)" -}} {{- range slice "object" "iframe" "video" "audio" "embed" -}} {{- $h = $h | replaceRE (printf `(?s)<%s\b.*?` . .) "" -}} {{- end -}} {{- $h = $h | replaceRE `]*>` "" | replaceRE `` "\n" -}} {{- /* Link and heading text on one line; a linked image keeps its alt text. */ -}} {{- range findRESubmatch `(?s)]*?href="([^"]*)"[^>]*>(.*?)` $h -}} {{- $text := index . 2 | replaceRE `]*?alt="([^"]*)"[^>]*>` " $1 " | plainify | htmlUnescape | replaceRE `\s+` " " | strings.TrimSpace -}} {{- $h = replace $h (index . 0) (cond (eq $text "") "" (printf "[%s](%s)" $text (index . 1))) -}} {{- end -}} {{- range findRESubmatch `(?s)]*>(.*?)` $h -}} {{- $text := index . 2 | plainify | htmlUnescape | replaceRE `\s+` " " | strings.TrimSpace -}} {{- $h = replace $h (index . 0) (printf "\n\n%s %s\n\n" (strings.Repeat (int (index . 1)) "#") $text) -}} {{- end -}} {{- $h = $h | replaceRE `]*?src="data:[^"]*"[^>]*?alt="([^"]*)"[^>]*>` "$1" -}} {{- $h = $h | replaceRE `]*?src="([^"]+)"[^>]*?alt="([^"]*)"[^>]*>` "![$2]($1)" -}} {{- $h = $h | replaceRE `]*?alt="([^"]*)"[^>]*?src="([^"]+)"[^>]*>` "![$1]($2)" -}} {{- $h = $h | replaceRE `]*?src="([^"]+)"[^>]*>` "![]($1)" -}} {{- $h = $h | replaceRE `(?s)<(?:strong|b)\b[^>]*>\s*(.*?)\s*` "**$1**" -}} {{- $h = $h | replaceRE `(?s)<(?:em|i)\b[^>]*>\s*(.*?)\s*` "_${1}_" -}} {{- return $h -}}
